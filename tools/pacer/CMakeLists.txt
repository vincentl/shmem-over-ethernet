# osoe-pacer: puts the library through its paces - whole-job patterns from real
# OpenSHMEM codes, every result checked against an oracle
# (docs/design/osoe-pacer-proposal.md). Installed beside oshcc and osoe-stat,
# so that `osoe-pacer --quick` under a cluster's launcher says whether the
# library works there; it finds the library from where it is, as oshcc does,
# so the prefix may still be chosen at install time. base/size.c is compiled
# in so that --memory and the heap's size parse exactly as the library parses
# SHMEM_SYMMETRIC_SIZE; libosoe hides its own copy.
set(osoe_pacer_sources osoe-pacer.c parallel-print.c sort.c alltoall.c mailbox.c teams.c
  threads.c print.c atomics.c rma.c locks.c memory.c collectives.c waits.c deprecated.c
  library-counters.c
  ${PROJECT_SOURCE_DIR}/base/size.c ${PROJECT_SOURCE_DIR}/tools/stat/segment.c)
add_executable(osoe-pacer ${osoe_pacer_sources})
target_include_directories(osoe-pacer PRIVATE ${PROJECT_SOURCE_DIR} ${PROJECT_SOURCE_DIR}/tools/stat)
find_package(Threads REQUIRED)
target_link_libraries(osoe-pacer PRIVATE osoe Threads::Threads m)
osoe_warnings(osoe-pacer)
set_target_properties(osoe-pacer PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
install(TARGETS osoe-pacer)

# osoe-pacer-counted: the same program with a shim that counts every call of
# every routine the library declares (spec §10: every shmem_ name is a weak
# alias of its pshmem_ routine, so a strong definition in the program counts
# and forwards). generate-shim.py writes the shim from the preprocessor's
# expansion of pshmem.h, whose declarations come from X-macro lists; nothing
# in the generated file is edited by hand. --coverage FILE writes the report.
# osoe-pacer-start-pes-counted is the Annex F program the same way.
#
# Python 3 writes the shim, and nothing else in the build needs it but the
# models (tools/modelcheck/CMakeLists.txt). So it is optional, found or not as
# for the models: where it is found these two programs, and the first's test,
# are built, and where it is not they are left out, with a status message. A
# machine without Python still builds, installs and tests everything else.
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/pshmem-preprocessed.i
    COMMAND ${CMAKE_C_COMPILER} -E -P -I${PROJECT_SOURCE_DIR}/shmem/include
            ${PROJECT_SOURCE_DIR}/shmem/include/pshmem.h
            -o ${CMAKE_CURRENT_BINARY_DIR}/pshmem-preprocessed.i
    DEPENDS ${PROJECT_SOURCE_DIR}/shmem/include/pshmem.h
            ${PROJECT_SOURCE_DIR}/shmem/include/shmem.h
    COMMENT "pacer: preprocessing pshmem.h for the counting shim"
    VERBATIM)
  add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/pacer-shim.c
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate-shim.py
            ${CMAKE_CURRENT_BINARY_DIR}/pshmem-preprocessed.i
            ${CMAKE_CURRENT_BINARY_DIR}/pacer-shim.c
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/generate-shim.py
            ${CMAKE_CURRENT_BINARY_DIR}/pshmem-preprocessed.i
    COMMENT "pacer: generating the call-counting shim"
    VERBATIM)
  # Two programs compile the shim. A custom command's output that is a source
  # of two targets may be generated by both at once under a parallel build -
  # CMake's add_custom_command says so, and a fresh -j4 build failed once on
  # it - so one target owns the generation and both programs wait for it.
  add_custom_target(osoe-pacer-shim DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/pacer-shim.c)
  add_executable(osoe-pacer-start-pes-counted start-pes.c
    ${CMAKE_CURRENT_BINARY_DIR}/pacer-shim.c)
  add_dependencies(osoe-pacer-start-pes-counted osoe-pacer-shim)
  target_link_libraries(osoe-pacer-start-pes-counted PRIVATE osoe)
  target_compile_definitions(osoe-pacer-start-pes-counted PRIVATE PACER_COUNTED=1)
  osoe_warnings(osoe-pacer-start-pes-counted)

  # internode: its one test runs over two nodes
  if(OSOE_INTERNODE)
    add_executable(osoe-pacer-counted ${osoe_pacer_sources}
      ${CMAKE_CURRENT_BINARY_DIR}/pacer-shim.c)
    add_dependencies(osoe-pacer-counted osoe-pacer-shim)
    target_include_directories(osoe-pacer-counted PRIVATE ${PROJECT_SOURCE_DIR}
      ${PROJECT_SOURCE_DIR}/tools/stat)
    target_link_libraries(osoe-pacer-counted PRIVATE osoe Threads::Threads m)
    target_compile_definitions(osoe-pacer-counted PRIVATE PACER_COUNTED=1)
    osoe_warnings(osoe-pacer-counted)
  endif()
else()
  message(STATUS "python3 not found: osoe-pacer-counted and osoe-pacer-start-pes-counted "
                 "will not be built, nor their test run")
endif()

# mix's end agreed: the time is up when the PE that has run longest says so.
# Each PE once read its own clock at the top of a period, and a run whose
# periods ended just at --seconds split - a PE that read the time as up left
# for the teardown's world sync while its team's other member began the next
# period, and the job hung (2026-10-08: a 60 s run, whose seed's first five
# periods took 60.0 s, about one run in forty). Here PE 1's clock starts an
# hour late, so that by its own clock its time is never up: PEs each deciding
# for itself split at the first period's end past 4 s, however long the build's
# periods take, and the deadlock detector ends the job. A late start of 2 s split
# them only where that end came within 2 s of the deadline, which a debug run
# met by half a second (the second adversarial read of Part 3, S3); agreed, the
# longest-run PE's clock decides, and PE 1's late one changes nothing.
add_test(NAME tools.pacer.mix-end.0-0-0-0
  COMMAND osoe-launch -n 4 -m 0,0,0,0 -t 60 -- $<TARGET_FILE:osoe-pacer> --seconds 4

  foreach(defect_line IN ITEMS
      "my-pe|start-pes: PE 1: _my_pe says 2 and _num_pes 4, where shmem_my_pe says 1"
      "put|start-pes: PE 2: word 32 of the shmalloc'd block holds [0-9]+ from PE 1"
      "realloc|start-pes: PE 1: shrealloc to [0-9]+ bytes changed word 0, [0-9]+, expected [0-9]+"
      "align|start-pes: PE 1: shmemalign.64, 100. returned 0x[0-9a-f]+, not a multiple of 64")
    string(FIND "${defect_line}" "|" bar)
    string(SUBSTRING "${defect_line}" 0 ${bar} defect)
    math(EXPR after "${bar} + 1")
    string(SUBSTRING "${defect_line}" ${after} -1 line)
    set(test tools.pacer.plant.start-pes-${defect})
    if(nodes EQUAL 1)
      set(test ${test}.${label})
    endif()

      COMMAND osoe-launch -k -n ${count} -m ${map} -t 240 -- $<TARGET_FILE:osoe-pacer-start-pes>
              --plant ${defect})
    set_tests_properties(${test} PROPERTIES LABELS "tier2"
      TIMEOUT 300 PASS_REGULAR_EXPRESSION "${line}.*start-pes failures=[1-9][0-9]* plant=${defect}"
      FAIL_REGULAR_EXPRESSION "killed by signal|timeout after|left behind")
    if(nodes GREATER 1)
      osoe_internode_label(${test})
    endif()
  endforeach()
endforeach()

# The planted defects: each check must still fire - the brief's four, and
# three more the author added on 2026-09-27 so that every check has one (the
# capacity plant loses keys, so the oracle fires beside it). Every line a
# plant's checks must print is required, each on its own, by
# cmake/PacerPlant.cmake: several PEs' lines come in an order that depends
# on who reserved first, and one pattern of "line, then count" would let
# every check but the first go quiet unnoticed. The alltoall's totals say
# that all eight variants caught it, every exchange. A `.` stands for each
# `;` of a line, which would otherwise split the list of patterns.
# Every planted run also says so, in a line after the header and on its
# last line, so that it cannot be taken for a finding.
#
# On one node every plant fires, by every line its test requires, but one:
# mailbox-round-trip changes a message of PE 1's round trips with its partner,
# the PE of its rank on the other node of its pair, and a job of one node has no
# such PE - osoe-pacer refuses the run, status 2. The round trips are the
# mailbox's check across nodes alone (a node-mate's message is timed on one
# clock instead, mailbox.c), so their plant is registered across nodes alone.
set(pacer_plants_across_nodes_only mailbox-round-trip)
function(pacer_plant_named name defect exercise)
  set(patterns "osoe-pacer planted ${defect}: PE 1[ ']"
      "osoe-pacer failures=[1-9][0-9]* seconds=[0-9.]+ plant=${defect}" ${ARGN})
  foreach(map IN LISTS pacer_plant_maps)
    osoe_map_nodes("${map}" nodes)
    string(REPLACE "," "-" label "${map}")
    set(test tools.pacer.plant.${name})
    if(nodes EQUAL 1)
      if("${name}" IN_LIST pacer_plants_across_nodes_only)
        continue()
      endif()
      set(test ${test}.${label})
    endif()
    add_test(NAME ${test}
      COMMAND ${CMAKE_COMMAND} -DLAUNCH=$<TARGET_FILE:osoe-launch>
              -DPACER=$<TARGET_FILE:osoe-pacer> -DMAP=${map}
              "-DARGUMENTS=--quick;--plant;${defect};${exercise}" "-DPATTERNS=${patterns}"
              "-DFAIL=killed by signal|timeout after|left behind"
              -P ${PROJECT_SOURCE_DIR}/cmake/PacerPlant.cmake)
    set_tests_properties(${test} PROPERTIES LABELS "tier2" TIMEOUT 300)
    if(nodes GREATER 1)

# threads, under SHMEM_THREAD_MULTIPLE, with both of the library's test hooks
# and a landing zone its gets overfill, as the library's own checked threaded
# test runs (shmem/tests/CMakeLists.txt): OSOE_CHECK_QUIET checks every shared
# wait's promise at its return, and OSOE_TEST_PERTURB widens the windows between
# a protocol's steps. On the four maps and on 0,1,2,0,1,0, where the hook's own
# race showed (docs/decisions/2026-09-27.md item 10); under tsan on 0,1 and on
# 0, one PE a node, where every writer of a PE's memory is a thread of the
# process tsan sees - 0 so that the single-node build runs it there too (the
# v0.2 step 1 review, correctness K1).
#
# Its collectives phase, in which each thread leads sums, scans and broadcasts
# on teams of its own (T25; round six's H1), is why 0,1,2,3 and 0,0,1,1,2,2 are
# here: the maps of the library's shmem.lanes.threads and its two-a-node twin,
# one PE a logical node, where every PE is a lead, and two a node, where a node
# has a lead and a PE that is not. A lane shared by threads - own_lane in
# internode-generic.c returning 0, or the scans' lane bound to 0 - passed
# the pacer's other exercises, mix, collectives and teams among them, because
# none of them has two threads of a PE lead collectives at once (the review; and
# the rest of this file's tests pass under either mutant, 2026-10-20). These
# fail it: 20 runs of 20 on each of the five maps of more than one node, for
# each mutant, the first by the library's own message that a piece is another
# collective's or nothing, the second by wrong prefix sums with status 0, which
# the team's scans alone missed in 3 runs of 25 and the copy's large scans do not.
# 0,0,1,1,2,2 is not run under tsan, as shmem.lanes.threads.two-a-node is not: a
# lead's agent writes its node-mate's heap from another process, which tsan
# cannot see ordered.
set(pacer_threads_maps 0,0,0,0 0,1,2,3 0,0,0,1,1,2 0,1,0,1 0,1,2,0,1,0 0,0,1,1,2,2)
if(CMAKE_C_FLAGS MATCHES "sanitize=thread")
  set(pacer_threads_maps 0,1 0)
endif()
foreach(map IN LISTS pacer_threads_maps)
  string(REPLACE "," ";" ranks "${map}")
  list(LENGTH ranks count)
  string(REPLACE "," "-" label "${map}")
  osoe_map_nodes("${map}" nodes)
  if(nodes GREATER 1 AND NOT OSOE_INTERNODE)
    continue()
  endif()
  add_test(NAME tools.pacer.threads.${label}
    COMMAND ${CMAKE_COMMAND} -E env OSOE_CHECK_QUIET=1 OSOE_TEST_PERTURB=1
            OSOE_GET_LANDING_BYTES=8192
            $<TARGET_FILE:osoe-launch> -n ${count} -m ${map} -t 240 --
            $<TARGET_FILE:osoe-pacer> --quick threads)
  set_tests_properties(tools.pacer.threads.${label} PROPERTIES LABELS "tier2" TIMEOUT 300
    PASS_REGULAR_EXPRESSION "osoe-pacer failures=0 "
    FAIL_REGULAR_EXPRESSION "${pacer_failed}|OSOE_CHECK_QUIET:")
  if(nodes GREATER 1)
    osoe_internode_label(tools.pacer.threads.${label})
  endif()
endforeach()
